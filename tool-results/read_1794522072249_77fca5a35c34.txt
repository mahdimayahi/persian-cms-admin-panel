   158→/** تعداد کارهای منتظر کاربر برای نشان منوی «کارهای من» */
   159→type TaskBadgeCounts = { awaitingReview: number; openTasks: number; overdueReviews: number }
   160→
   161→export function useTaskBadge() {
   162→  const [counts, setCounts] = React.useState<TaskBadgeCounts | null>(null)
   163→  const load = React.useCallback(async () => {
   164→    try {
   165→      const res = await apiGet<TaskBadgeCounts & Record<string, unknown>>('/api/tasks')
   166→      setCounts({
   167→        awaitingReview: (res as unknown as { counts?: TaskBadgeCounts }).counts?.awaitingReview ?? 0,
   168→        openTasks: (res as unknown as { counts?: TaskBadgeCounts }).counts?.openTasks ?? 0,
   169→        overdueReviews: (res as unknown as { counts?: TaskBadgeCounts }).counts?.overdueReviews ?? 0,
   170→      })
   171→    } catch {
   172→      // بی‌صدا: نشان حیاتی نیست
   173→    }
   174→  }, [])
   175→  React.useEffect(() => {
   176→    load()
   177→    const t = setInterval(load, 120_000)
   178→    return () => clearInterval(t)
   179→  }, [load])
   180→  return counts
   181→}

   345→/* ─── اعلان‌ها ───────────────────────────────────────────────── */
   346→function NotificationsBell() {
   347→  const [open, setOpen] = React.useState(false)
   348→  const [items, setItems] = React.useState<NotificationItem[]>([])
   349→  const [loading, setLoading] = React.useState(false)
   350→  const unreadCount = useAppStore((s) => s.unreadCount)
   351→  const setUnreadCount = useAppStore((s) => s.setUnreadCount)
   352→
   353→  const load = React.useCallback(async () => {
   354→    setLoading(true)
   355→    try {
   356→      const res = await apiGet<{ items: NotificationItem[]; unreadCount: number }>('/api/notifications?pageSize=15')
   357→      setItems(res.items)
   358→      setUnreadCount(res.unreadCount)
   359→    } catch {
   360→      // بی‌صدا: اعلان حیاتی نیست
   361→    } finally {
   362→      setLoading(false)
   363→    }
   364→  }, [setUnreadCount])
   365→
   366→  React.useEffect(() => {
   367→    load()
   368→    const t = setInterval(load, 30000)
   369→    return () => clearInterval(t)
   370→  }, [load])
   371→
   372→  const markRead = async (id?: string) => {
   373→    try {
   374→      await apiPut('/api/notifications/read', id ? { ids: [id] } : { all: true })
   375→      await load()
   376→    } catch {
   377→      toast.error('ثبت وضعیت اعلان ناموفق بود.')
   378→    }
   379→  }
   380→
   381→  return (
   382→    <Popover open={open} onOpenChange={setOpen}>
   383→      <PopoverTrigger asChild>
   384→        <Button
   385→          variant="ghost"
   386→          size="icon"
   387→          className="relative rounded-xl hover:bg-muted"
   388→          aria-label={`اعلان‌ها${unreadCount ? `، ${toFa(unreadCount)} خوانده‌نشده` : ''}`}
   389→        >
   390→          <Bell className="h-5 w-5" />
   391→          {unreadCount > 0 ? (
   392→            <span className="absolute -top-0.5 -end-0.5 h-4.5 min-w-4.5 rounded-full bg-destructive text-destructive-foreground text-[10px] leading-4 flex items-center justify-center px-1">
   393→              {unreadCount > 9 ? '۹+' : toFa(unreadCount)}
   394→            </span>
   395→          ) : null}
   396→        </Button>
   397→      </PopoverTrigger>
   398→      <PopoverContent align="end" className="w-96 p-0 rounded-2xl" dir="rtl">
   399→        <div className="flex items-center justify-between px-4 py-3 border-b">
   400→          <p className="font-semibold text-sm flex items-center gap-2">
   401→            اعلان‌ها
   402→            {unreadCount > 0 ? (
   403→              <span className="rounded-full bg-primary/10 text-primary text-[10px] px-2 py-0.5 font-semibold">
   404→                {toFa(unreadCount)} جدید
   405→              </span>
   406→            ) : null}
   407→          </p>
   408→          {unreadCount > 0 ? (
   409→            <Button variant="ghost" size="sm" className="text-xs h-7" onClick={() => markRead()}>
   410→              خواندن همه
   411→            </Button>
   412→          ) : null}
   413→        </div>
   414→        <ScrollArea className="h-80">
   415→          {loading && items.length === 0 ? (
   416→            <div className="p-4 space-y-3">
   417→              {Array.from({ length: 3 }).map((_, i) => (
   418→                <Skeleton key={i} className="h-12 w-full rounded-xl" />
   419→              ))}
   420→            </div>
   421→          ) : items.length === 0 ? (
   422→            <p className="text-sm text-muted-foreground text-center py-10">اعلانی وجود ندارد.</p>
   423→          ) : (
   424→            <ul className="divide-y">
   425→              {items.map((n) => (
   426→                <li key={n.id}>
   427→                  <button
   428→                    type="button"
   429→                    className={cn('w-full text-start px-4 py-3 hover:bg-muted/60 transition', !n.read && 'bg-primary/5')}
   430→                    onClick={() => {
   431→                      if (!n.read) markRead(n.id)
   432→                      setOpen(false)
   433→                      if (n.link) window.location.hash = n.link.replace(/^#/, '')
   434→                    }}
   435→                  >
   436→                    <div className="flex items-start justify-between gap-2">
   437→                      <p className={cn('text-sm leading-5', !n.read && 'font-semibold')}>{n.title}</p>
   438→                      {!n.read ? <span className="h-2 w-2 rounded-full bg-primary mt-1.5 shrink-0" aria-label="خوانده‌نشده" /> : null}
   439→                    </div>
   440→                    {n.body ? <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{n.body}</p> : null}
   441→                    <p className="text-[11px] text-muted-foreground mt-1">{timeAgo(n.createdAt)}</p>
   442→                  </button>
   443→                </li>
   444→              ))}
   445→            </ul>
   446→          )}
   447→        </ScrollArea>
   448→      </PopoverContent>
   449→    </Popover>
   450→  )
   451→}

   475→function ThemeToggle() {
   476→  const { theme, setTheme } = useTheme()
   477→  const [mounted, setMounted] = React.useState(false)
   478→  React.useEffect(() => setMounted(true), [])
   479→  return (
   480→    <Button
   481→      variant="ghost"
   482→      size="icon"
   483→      className="rounded-xl hover:bg-muted"
   484→      onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   485→      aria-label={mounted && theme === 'dark' ? 'تغییر به حالت روشن' : 'تغییر به حالت تیره'}
   486→    >
   487→      {mounted && theme === 'dark' ? <Sun className="h-5 w-5" /> : <Moon className="h-5 w-5" />}
   488→    </Button>
   489→  )
   490→}

   494→  const user = useAppStore((s) => s.user)
   495→  const logout = useAppStore((s) => s.logout)
   496→  if (!user) return null
   497→  const doLogout = async () => {
   498→    try {
   499→      await apiPost('/api/auth/logout')
   500→    } catch {
   501→      // حتی در صورت خطا، از پنل خارج می‌شویم
   502→    }
   503→    logout()
   504→    toast.success('با موفقیت خارج شدید.')
   505→  }
   506→  return (
   507→    <DropdownMenu>
   508→      <DropdownMenuTrigger asChild>
   509→        <button
   510→          type="button"
   511→          className="flex items-center gap-2 rounded-xl outline-none focus-visible:ring-2 focus-visible:ring-ring hover:bg-muted p-1 ps-2 transition-colors"
   512→          aria-label="منوی حساب کاربری"
   513→        >
   514→          <Avatar className="h-8 w-8 ring-2 ring-primary/20">
   515→            <AvatarFallback className="bg-primary text-primary-foreground text-xs font-bold">
   516→              {initials(user.name)}
   517→            </AvatarFallback>
   518→          </Avatar>
   519→          <span className="hidden lg:inline text-sm font-medium">{user.name}</span>
   520→          <ChevronDown className="h-3.5 w-3.5 text-muted-foreground hidden lg:inline" aria-hidden />
   521→        </button>
   522→      </DropdownMenuTrigger>
   523→      <DropdownMenuContent align="end" className="w-56 rounded-xl">
   524→        <DropdownMenuLabel>
   525→          <p className="font-semibold">{user.name}</p>
   526→          <p className="text-xs text-muted-foreground font-normal text-start" dir="ltr">
   527→            {user.email}
   528→          </p>
   529→          <Badge variant="secondary" className="mt-1.5 pointer-events-none bg-primary/10 text-primary">
   530→            {ROLE_LABELS[user.role as Role] ?? user.role}
   531→          </Badge>
   532→        </DropdownMenuLabel>
   533→        <DropdownMenuSeparator />
   534→        <DropdownMenuItem onClick={() => (window.location.hash = '/account')}>
   535→          <UserRound className="h-4 w-4 me-2" />
   536→          پروفایل و امنیت
   537→        </DropdownMenuItem>
   538→        <DropdownMenuItem onClick={onReport}>
   539→          <LifeBuoy className="h-4 w-4 me-2" />
   540→          گزارش مشکل
   541→        </DropdownMenuItem>
   542→        <DropdownMenuItem variant="destructive" onClick={doLogout}>
   543→          <LogOut className="h-4 w-4 me-2" />
   544→          خروج از حساب
   545→        </DropdownMenuItem>
   546→      </DropdownMenuContent>
   547→    </DropdownMenu>
   548→  )
   549→}

   558→  navigate: (to: string) => void

   561→  const [mobileOpen, setMobileOpen] = React.useState(false)
   562→  const [paletteOpen, setPaletteOpen] = React.useState(false)
   563→  const [supportOpen, setSupportOpen] = React.useState(false)
   564→  const [sitePreviewOpen, setSitePreviewOpen] = React.useState(false)
   565→  const taskBadge = useTaskBadge()
   566→
   567→  // میان‌بر Ctrl+K / Cmd+K برای پالت فرمان

   570→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {

   572→        setPaletteOpen((o) => !o)

   576→    return () => window.removeEventListener('keydown', onKey)
   577→  }, [])
   578→

   580→  const crumb = BREADCRUMBS[base] ?? 'پنل مدیریت'

   582→    '/articles/new': 'مقاله جدید',
   583→    '/account/security': 'امنیت',
   584→  }
   585→  const subCrumb =
   586→    path.startsWith('/articles/') && path !== '/articles'
   587→      ? path === '/articles/new'
   588→        ? 'مقاله جدید'
