   278→function NotificationsBell() {
   279→  const [open, setOpen] = React.useState(false)
   280→  const [items, setItems] = React.useState<NotificationItem[]>([])
   281→  const [loading, setLoading] = React.useState(false)
   282→  const unreadCount = useAppStore((s) => s.unreadCount)
   283→  const setUnreadCount = useAppStore((s) => s.setUnreadCount)
   284→
   285→  const load = React.useCallback(async () => {
   286→    setLoading(true)
   287→    try {
   288→      const res = await apiGet<{ items: NotificationItem[]; unreadCount: number }>('/api/notifications?pageSize=15')
   289→      setItems(res.items)
   290→      setUnreadCount(res.unreadCount)
   291→    } catch {
   292→      // بی‌صدا: اعلان حیاتی نیست
   293→    } finally {
   294→      setLoading(false)
   295→    }
   296→  }, [setUnreadCount])
   297→
   298→  React.useEffect(() => {
   299→    load()
   300→    const t = setInterval(load, 30000)
   301→    return () => clearInterval(t)
   302→  }, [load])
   303→
   304→  const markRead = async (id?: string) => {
   305→    try {
   306→      await apiPut('/api/notifications/read', id ? { ids: [id] } : { all: true })
   307→      await load()
   308→    } catch {
   309→      toast.error('ثبت وضعیت اعلان ناموفق بود.')
   310→    }
   311→  }
   312→
   313→  return (
   314→    <Popover open={open} onOpenChange={setOpen}>
   315→      <PopoverTrigger asChild>
   316→        <Button
   317→          variant="ghost"
   318→          size="icon"
   319→          className="relative rounded-xl hover:bg-muted"
   320→          aria-label={`اعلان‌ها${unreadCount ? `، ${toFa(unreadCount)} خوانده‌نشده` : ''}`}
   321→        >
   322→          <Bell className="h-5 w-5" />
   323→          {unreadCount > 0 ? (
   324→            <span className="absolute -top-0.5 -end-0.5 h-4.5 min-w-4.5 rounded-full bg-gradient-to-br from-destructive to-chart-4 text-[10px] leading-4 text-white flex items-center justify-center px-1 shadow-sm">
   325→              {unreadCount > 9 ? '۹+' : toFa(unreadCount)}
   326→            </span>
   327→          ) : null}
   328→        </Button>
   329→      </PopoverTrigger>
   330→      <PopoverContent align="end" className="w-96 p-0 rounded-2xl" dir="rtl">
   331→        <div className="flex items-center justify-between px-4 py-3 border-b">
   332→          <p className="font-semibold text-sm flex items-center gap-2">
   333→            اعلان‌ها
   334→            {unreadCount > 0 ? (
   335→              <span className="rounded-full bg-primary/10 text-primary text-[10px] px-2 py-0.5 font-semibold">
   336→                {toFa(unreadCount)} جدید
   337→              </span>
   338→            ) : null}
   339→          </p>
   340→          {unreadCount > 0 ? (
   341→            <Button variant="ghost" size="sm" className="text-xs h-7" onClick={() => markRead()}>
   342→              خواندن همه
   343→            </Button>
   344→          ) : null}
   345→        </div>
   346→        <ScrollArea className="h-80">
   347→          {loading && items.length === 0 ? (
   348→            <div className="p-4 space-y-3">
   349→              {Array.from({ length: 3 }).map((_, i) => (
   350→                <Skeleton key={i} className="h-12 w-full rounded-xl" />
   351→              ))}
   352→            </div>
   353→          ) : items.length === 0 ? (
   354→            <p className="text-sm text-muted-foreground text-center py-10">اعلانی وجود ندارد.</p>
   355→          ) : (
   356→            <ul className="divide-y">
   357→              {items.map((n) => (
   358→                <li key={n.id}>
   359→                  <button
   360→                    type="button"
   361→                    className={cn('w-full text-start px-4 py-3 hover:bg-muted/60 transition', !n.read && 'bg-primary/5')}
   362→                    onClick={() => {
   363→                      if (!n.read) markRead(n.id)
   364→                      setOpen(false)
   365→                      if (n.link) window.location.hash = n.link.replace(/^#/, '')
   366→                    }}
   367→                  >
   368→                    <div className="flex items-start justify-between gap-2">
   369→                      <p className={cn('text-sm leading-5', !n.read && 'font-semibold')}>{n.title}</p>
   370→                      {!n.read ? <span className="h-2 w-2 rounded-full bg-primary mt-1.5 shrink-0" aria-label="خوانده‌نشده" /> : null}
   371→                    </div>
   372→                    {n.body ? <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{n.body}</p> : null}
   373→                    <p className="text-[11px] text-muted-foreground mt-1">{timeAgo(n.createdAt)}</p>
   374→                  </button>
   375→                </li>
   376→              ))}
   377→            </ul>
   378→          )}
   379→        </ScrollArea>
   380→      </PopoverContent>
   381→    </Popover>
   382→  )
   383→}

   385→/* ─── جست‌وجوی سراسری ────────────────────────────────────────── */
   386→function GlobalSearch() {
   387→  const [q, setQ] = React.useState('')
   388→  const [results, setResults] = React.useState<SearchResult | null>(null)
   389→  const [open, setOpen] = React.useState(false)
   390→  const [loading, setLoading] = React.useState(false)
   391→  const boxRef = React.useRef<HTMLDivElement>(null)
   392→
   393→  React.useEffect(() => {
   394→    if (q.trim().length < 2) {
   395→      setResults(null)
   396→      setOpen(false)
   397→      return
   398→    }
   399→    const t = setTimeout(async () => {
   400→      setLoading(true)
   401→      try {
   402→        const res = await apiGet<SearchResult>(`/api/search?q=${encodeURIComponent(q.trim())}`)
   403→        setResults(res)
   404→        setOpen(true)
   405→      } catch {
   406→        setResults(null)
   407→      } finally {
   408→        setLoading(false)
   409→      }
   410→    }, 300)
   411→    return () => clearTimeout(t)
   412→  }, [q])
   413→
   414→  React.useEffect(() => {
   415→    const onDown = (e: MouseEvent) => {
   416→      if (boxRef.current && !boxRef.current.contains(e.target as Node)) setOpen(false)
   417→    }
   418→    document.addEventListener('mousedown', onDown)
   419→    return () => document.removeEventListener('mousedown', onDown)
   420→  }, [])
   421→
   422→  const go = (hash: string) => {
   423→    setOpen(false)
   424→    setQ('')
   425→    window.location.hash = hash.replace(/^#/, '')
   426→  }
   427→
   428→  const hasResults = results && results.articles.length + results.pages.length + results.media.length + results.users.length > 0
   429→
   430→  return (
   431→    <div ref={boxRef} className="relative w-full max-w-sm hidden md:block" role="search">
   432→      <Search className="absolute start-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground pointer-events-none" aria-hidden />
   433→      <Input
   434→        value={q}
   435→        onChange={(e) => setQ(e.target.value)}
   436→        onFocus={() => {
   437→          if (q.trim().length >= 2 && results) setOpen(true)
   438→        }}
   439→        placeholder="جست‌وجوی سراسری…"
   440→        className="ps-9 rounded-xl bg-muted/60 border-transparent focus-visible:bg-background focus-visible:border-ring h-9.5"
   441→        aria-label="جست‌وجوی سراسری"
   442→      />
   443→      {open ? (
   444→        <div className="absolute top-full mt-2 w-full rounded-2xl border bg-popover shadow-lg z-50 overflow-hidden">
   445→          {loading ? (
   446→            <p className="text-sm text-muted-foreground p-4">در حال جست‌وجو…</p>
   447→          ) : !hasResults ? (
   448→            <p className="text-sm text-muted-foreground p-4">نتیجه‌ای یافت نشد.</p>
   449→          ) : (
   450→            <ScrollArea className="max-h-80">
   451→              <div className="p-2 space-y-3">
   452→                {results!.articles.length > 0 ? (
   453→                  <div>
   454→                    <p className="text-xs font-medium text-muted-foreground px-2 py-1">مقالات</p>
   455→                    {results!.articles.map((a) => (
   456→                      <button
   457→                        key={a.id}
   458→                        type="button"
   459→                        onClick={() => go(`#/articles/${a.id}/edit`)}
   460→                        className="w-full text-start px-2 py-1.5 rounded-lg hover:bg-muted text-sm flex items-center justify-between gap-2"
   461→                      >
   462→                        <span className="truncate">{a.title}</span>
   463→                        <span className="text-[10px] text-muted-foreground shrink-0">مقاله</span>
   464→                      </button>
   465→                    ))}
   466→                  </div>
   467→                ) : null}
   468→                {results!.pages.length > 0 ? (
   469→                  <div>
   470→                    <p className="text-xs font-medium text-muted-foreground px-2 py-1">صفحات</p>
   471→                    {results!.pages.map((p) => (
   472→                      <button
   473→                        key={p.id}

   526→function ThemeToggle() {
   527→  const { theme, setTheme } = useTheme()
   528→  const [mounted, setMounted] = React.useState(false)
   529→  React.useEffect(() => setMounted(true), [])
   530→  return (
   531→    <Button
   532→      variant="ghost"
   533→      size="icon"
   534→      className="rounded-xl hover:bg-muted"
   535→      onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   536→      aria-label={mounted && theme === 'dark' ? 'تغییر به حالت روشن' : 'تغییر به حالت تیره'}
   537→    >
   538→      {mounted && theme === 'dark' ? <Sun className="h-5 w-5" /> : <Moon className="h-5 w-5" />}
   539→    </Button>
   540→  )
   541→}
