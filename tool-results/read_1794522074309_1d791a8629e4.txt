   159→   159→type TaskBadgeCounts = { awaitingReview: number; openTasks: number; overdueReviews: number }
   160→   160→
   161→   161→export function useTaskBadge() {
   162→   162→  const [counts, setCounts] = React.useState<TaskBadgeCounts | null>(null)
   163→   163→  const load = React.useCallback(async () => {
   164→   164→    try {
   165→   165→      const res = await apiGet<TaskBadgeCounts & Record<string, unknown>>('/api/tasks')
   166→   166→      setCounts({
   167→   167→        awaitingReview: (res as unknown as { counts?: TaskBadgeCounts }).counts?.awaitingReview ?? 0,
   168→   168→        openTasks: (res as unknown as { counts?: TaskBadgeCounts }).counts?.openTasks ?? 0,
   169→   169→        overdueReviews: (res as unknown as { counts?: TaskBadgeCounts }).counts?.overdueReviews ?? 0,
   170→   170→      })
   171→   171→    } catch {
   172→   172→      // بی‌صدا: نشان حیاتی نیست
   173→   173→    }
   174→   174→  }, [])
   175→   175→  React.useEffect(() => {
   176→   176→    load()
   177→   177→    const t = setInterval(load, 120_000)
   178→   178→    return () => clearInterval(t)
   179→   179→  }, [load])
   180→   180→  return counts
   181→   181→}
