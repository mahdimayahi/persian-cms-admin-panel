    56→    56→    56→async function apiPatch<T>(path: string, body?: unknown): Promise<T> {
    57→    57→    57→  const res = await fetch(path, {
    58→    58→    58→    method: 'PATCH',
    59→    59→    59→    headers: { 'Content-Type': 'application/json' },
    60→    60→    60→    body: body !== undefined ? JSON.stringify(body) : undefined,
    61→    61→    61→    credentials: 'same-origin',
    62→    62→    62→  })
    63→    63→    63→  if (!res.ok) {
    64→    64→    64→    const b = (await res.json().catch(() => ({}))) as { error?: string }
    65→    65→    65→    throw new Error(b.error || 'خطا')
    66→    66→    66→  }
    67→    67→    67→  return res.json() as Promise<T>
    68→    68→    68→}
