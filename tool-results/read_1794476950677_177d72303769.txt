    52→/**
    53→ * کمک‌تابع محلی برای endpointهایی که فقط PATCH را می‌پذیرند
    54→ * (کتابخانه api کلاینت متد PATCH ندارد) — با همان قرارداد خطای apiGet
    55→ */
    56→async function apiPatch<T>(path: string, body?: unknown): Promise<T> {
    57→  const res = await fetch(path, {
    58→    method: 'PATCH',
    59→    headers: { 'Content-Type': 'application/json' },
    60→    body: body !== undefined ? JSON.stringify(body) : undefined,
    61→    credentials: 'same-origin',
    62→  })
    63→  if (!res.ok) {
    64→    const b = (await res.json().catch(() => ({}))) as { error?: string }
    65→    throw new Error(b.error || 'خطا')
    66→  }
    67→  return res.json() as Promise<T>
    68→}
