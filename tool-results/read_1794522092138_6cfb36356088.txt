    10→function hashPassword(password: string): string {
    11→  const salt = randomBytes(16).toString('hex')
    12→  const hash = scryptSync(password, salt, 64).toString('hex')
    13→  return `${salt}:${hash}`
    14→}

    16→async function makeGradientPng(colors: [string, string]): Promise<string> {
    17→  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="960" height="600">
    18→    <defs><linearGradient id="g" x1="0" y1="0" x2="1" y2="1">
    19→      <stop offset="0%" stop-color="${colors[0]}"/><stop offset="100%" stop-color="${colors[1]}"/>
    20→    </linearGradient></defs>
    21→    <rect width="960" height="600" fill="url(#g)"/>
    22→    <circle cx="720" cy="140" r="90" fill="rgba(255,255,255,0.15)"/>
    23→    <circle cx="200" cy="460" r="130" fill="rgba(255,255,255,0.10)"/>
    24→  </svg>`
    25→  const sharp = (await import('sharp')).default
    26→  const buf = await sharp(Buffer.from(svg)).png().toBuffer()
    27→  return buf.toString('base64')
    28→}
